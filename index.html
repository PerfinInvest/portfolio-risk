        let symC = document.getElementById('symbol-c').value.trim().toUpperCase();


        // สุ่มหรือดึงค่า Correlation AB

        if (symA && symB) {

            let pairKey = `${symA}_${symB}`;

            let autoRhoAB = correlationMatrix[pairKey];

            if (autoRhoAB === undefined) {

                // คำนวณค่าประมาณสถิติตามลักษณะชื่อหุ้น

                let hash = 0;

                for (let i = 0; i < pairKey.length; i++) hash += pairKey.charCodeAt(i);

                autoRhoAB = parseFloat(((hash % 120 - 40) / 100).toFixed(2)); // ผลลัพธ์ช่วง -0.40 ถึง +0.80

            }

            document.getElementById('rho-ab').value = autoRhoAB;

        }


        // Correlation AC

        if (symA && symC) {

            let pairKey = `${symA}_${symC}`;

            let autoRhoAC = correlationMatrix[pairKey] || 0.45;

            document.getElementById('rho-ac').value = autoRhoAC;

        }


        // Correlation BC

        if (symB && symC) {

            let pairKey = `${symB}_${symC}`;

            let autoRhoBC = correlationMatrix[pairKey] || 0.35;

            document.getElementById('rho-bc').value = autoRhoBC;

        }

    }


    function fetchData(target) {

        let symInput = document.getElementById(`symbol-${target.toLowerCase()}`);

        let sdInput = document.getElementById(`sd-${target.toLowerCase()}`);

        let symbol = symInput.value.trim().toUpperCase();


        if (symbol === "") {

            if (target === 'C') {

                checkCSlot();

                calculateRisk();

            }

            return;

        }


        // ดึงค่า SD จาก DB หรือสุ่มตามความผันผวนสมจริง

        let fetchedSD = mockAssetDB[symbol];

        if (!fetchedSD) {

            let hash = 0;

            for (let i = 0; i < symbol.length; i++) hash += symbol.charCodeAt(i);

            fetchedSD = (14 + (hash % 25)).toFixed(1);

        }


        sdInput.value = fetchedSD;


        // คำนวณค่า Correlation ของคู่หุ้นให้อัตโนมัติทันที

        autoCalculateCorrelation();

        checkCSlot();

        calculateRisk();

    }


    function checkCSlot() {

        let symC = document.getElementById('symbol-c').value.trim();

        let boxAC = document.getElementById('box-rho-ac');

        let boxBC = document.getElementById('box-rho-bc');


        if (symC !== "") {

            boxAC.style.display = 'block';

            boxBC.style.display = 'block';

        } else {

            boxAC.style.display = 'none';

            boxBC.style.display = 'none';

        }

    }


    function calculateRisk() {

        let symA = document.getElementById('symbol-a').value.trim().toUpperCase() || 'หุ้น A';

        let symB = document.getElementById('symbol-b').value.trim().toUpperCase() || 'หุ้น B';

        let symC = document.getElementById('symbol-c').value.trim().toUpperCase();


        let amtA = parseFloat(document.getElementById('amt-a').value) || 0;

        let amtB = parseFloat(document.getElementById('amt-b').value) || 0;

        let amtC = parseFloat(document.getElementById('amt-c').value) || 0;


        let sdA = parseFloat(document.getElementById('sd-a').value) || 0;

        let sdB = parseFloat(document.getElementById('sd-b').value) || 0;

        let sdC = parseFloat(document.getElementById('sd-c').value) || 0;


        let hasC = symC !== "" && amtC > 0 && sdC > 0;


        let totalAmt = amtA + amtB + (hasC ? amtC : 0);

        let wA = totalAmt > 0 ? (amtA / totalAmt) : 0;

        let wB = totalAmt > 0 ? (amtB / totalAmt) : 0;

        let wC = (hasC && totalAmt > 0) ? (amtC / totalAmt) : 0;


        let rhoAB = parseFloat(document.getElementById('rho-ab').value) || 0;

        let rhoAC = parseFloat(document.getElementById('rho-ac').value) || 0;

        let rhoBC = parseFloat(document.getElementById('rho-bc').value) || 0;


        // Update Slider Labels

        document.getElementById('label-rho-ab').innerText = `${symA} กับ ${symB}`;

        document.getElementById('rho-ab-val').innerText = (rhoAB >= 0 ? '+' : '') + rhoAB.toFixed(2);

        

        if (hasC) {

            document.getElementById('label-rho-ac').innerText = `${symA} กับ ${symC}`;

            document.getElementById('rho-ac-val').innerText = (rhoAC >= 0 ? '+' : '') + rhoAC.toFixed(2);


            document.getElementById('label-rho-bc').innerText = `${symB} กับ ${symC}`;

            document.getElementById('rho-bc-val').innerText = (rhoBC >= 0 ? '+' : '') + rhoBC.toFixed(2);

        }


        // Render Summary Table

        let tableBody = document.getElementById('weight-table-body');

        let html = `

            <tr>

                <td><strong>${symA}</strong></td>

                <td>${amtA.toLocaleString()} บาท</td>

                <td>${(wA * 100).toFixed(1)}%</td>

                <td>${sdA.toFixed(1)}%</td>

            </tr>

            <tr>

                <td><strong>${symB}</strong></td>

                <td>${amtB.toLocaleString()} บาท</td>

                <td>${(wB * 100).toFixed(1)}%</td>

                <td>${sdB.toFixed(1)}%</td>

            </tr>

        `;


        if (hasC) {

            html += `

                <tr>

                    <td><strong>${symC}</strong></td>

                    <td>${amtC.toLocaleString()} บาท</td>

                    <td>${(wC * 100).toFixed(1)}%</td>

                    <td>${sdC.toFixed(1)}%</td>

                </tr>

            `;

        }


        html += `

            <tr style="background: #F9FAFB; font-weight: 700;">

                <td>รวมทั้งหมด</td>

                <td>${totalAmt.toLocaleString()} บาท</td>

                <td>100.0%</td>

                <td>-</td>

            </tr>

        `;

        tableBody.innerHTML = html;


        // Portfolio Variance Calculation

        let varA = Math.pow(wA, 2) * Math.pow(sdA, 2);

        let varB = Math.pow(wB, 2) * Math.pow(sdB, 2);

        let covAB = 2 * wA * wB * sdA * sdB * rhoAB;


        let totalVar = varA + varB + covAB;


        if (hasC) {

            let varC = Math.pow(wC, 2) * Math.pow(sdC, 2);

            let covAC = 2 * wA * wC * sdA * sdC * rhoAC;

            let covBC = 2 * wB * wC * sdB * sdC * rhoBC;

            totalVar += varC + covAC + covBC;

        }


        let portSD = Math.sqrt(Math.max(0, totalVar));


        document.getElementById('total-sd').innerText = portSD.toFixed(1) + '%';


        // Badge & Summary Logic

        let badge = document.getElementById('risk-badge');

        let summaryText = document.getElementById('summary-text');


        if (portSD < 15) {

            badge.className = "risk-badge badge-low";

            badge.innerText = "ความเสี่ยงต่ำ (LOW RISK)";

            summaryText.innerHTML = `<strong>พอร์ตผันผวนต่ำ:</strong> สินทรัพย์คู่นี้กระจายความเสี่ยงได้ดีเนื่องจากค่า Correlation (${rhoAB >= 0 ? '+' : ''}${rhoAB.toFixed(2)}) ช่วยลดความผันผวนรวมลง`;

        } else if (portSD <= 25) {

            badge.className = "risk-badge badge-mid";

            badge.innerText = "ความเสี่ยงปานกลาง (MODERATE RISK)";

            summaryText.innerHTML = `<strong>พอร์ตผันผวนปานกลาง:</strong> การจัดสัดส่วนการลงทุนและสถิติความสัมพันธ์อยู่ในระดับสมดุล เหมาะสมสำหรับการลงทุนระยะยาว`;

        } else {

            badge.className = "risk-badge badge-high";

            badge.innerText = "ความเสี่ยงสูง (HIGH RISK)";

            summaryText.innerHTML = `<strong>พอร์ตผันผวนสูง:</strong> สินทรัพย์มีความผันผวนสูง หรือเคลื่อนไหวตามกันมาก (Correlation = ${rhoAB >= 0 ? '+' : ''}${rhoAB.toFixed(2)}) แนะนำให้สลับไปถือสินทรัพย์ความเสี่ยงต่ำเพิ่มเติม`;

        }


        // Chart Update

        if (setChart) {

            let labels = hasC ? [symA, symB, symC, 'พอร์ตรวม'] : [symA, symB, 'พอร์ตรวม'];

            let data = hasC ? [sdA, sdB, sdC, portSD.toFixed(1)] : [sdA, sdB, portSD.toFixed(1)];

            

            setChart.data.labels = labels;

            setChart.data.datasets[0].data = data;

            setChart.data.datasets[0].backgroundColor = labels.map((l, i) => i === labels.length - 1 ? '#F37021' : '#9CA3AF');

            setChart.data.datasets[0].borderColor = labels.map((l, i) => i === labels.length - 1 ? '#D95D12' : '#4B5563');

            setChart.update();

        }

    }


    window.onload = function() {

        initChart();

        checkCSlot();

        calculateRisk();

    };
